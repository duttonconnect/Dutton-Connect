rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // -------------------------------------------------------------------------
    // HELPERS
    // -------------------------------------------------------------------------

    function isSignedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    // Reads the caller's own user doc to check the isAdmin flag.
    // For high-scale apps, replace with a Firebase Auth custom claim
    // (request.auth.token.isAdmin == true) to avoid the extra read.
    function isAdmin() {
      return isSignedIn() &&
        get(/databases/$(database)/documents/users/$(request.auth.uid))
          .data.get('isAdmin', false) == true;
    }


    // -------------------------------------------------------------------------
    // users/{uid}
    // -------------------------------------------------------------------------

    match /users/{uid} {

      // Any signed-in user can read any user profile.
      // NOTE: this exposes the email field to other authenticated users.
      // Future fix: move PII into users/{uid}/private/pii sub-document.
      allow read: if isSignedIn();

      // Users may only create their own document (signup flow).
      allow create: if isOwner(uid);

      // Users may update their own document but cannot grant themselves
      // the isAdmin flag. Admins may update any user document.
      allow update: if
        ( isOwner(uid) &&
          !request.resource.data.diff(resource.data)
            .affectedKeys().hasAny(['isAdmin']) )
        || isAdmin();

      // Only admins may delete user documents.
      allow delete: if isAdmin();

      // Private sub-collection used by FirestoreSyncBridge.
      // Only the owning user may read or write their own app state.
      match /private/{docId} {
        allow read, write: if isOwner(uid);
      }

    }


    // -------------------------------------------------------------------------
    // jobRequests/{id}
    // -------------------------------------------------------------------------

    match /jobRequests/{requestId} {

      // Open requests are visible to all signed-in users (pros browse them).
      // Non-open requests are visible only to the posting customer,
      // the accepted pro, or an admin.
      allow read: if isSignedIn() && (
        resource.data.status == 'open' ||
        resource.data.customerId == request.auth.uid ||
        resource.data.acceptedProId == request.auth.uid ||
        isAdmin()
      );

      // Only the customer may create a request; their uid must be the customerId.
      allow create: if isSignedIn() &&
        request.resource.data.customerId == request.auth.uid;

      // The posting customer (accepts quotes, schedules) or the accepted pro
      // (marks completed) may update. Admins may update anything.
      allow update: if isSignedIn() && (
        resource.data.customerId == request.auth.uid ||
        resource.data.acceptedProId == request.auth.uid ||
        isAdmin()
      );

      // Only the posting customer or an admin may delete.
      allow delete: if isSignedIn() && (
        resource.data.customerId == request.auth.uid ||
        isAdmin()
      );

    }


    // -------------------------------------------------------------------------
    // matchQuotes/{id}
    // -------------------------------------------------------------------------

    match /matchQuotes/{quoteId} {

      // Only the involved customer, the pro who sent the quote, or an admin
      // may read a quote.
      allow read: if isSignedIn() && (
        resource.data.customerId == request.auth.uid ||
        resource.data.proId == request.auth.uid ||
        isAdmin()
      );

      // Only pros may create quotes; proId must equal their own uid.
      allow create: if isSignedIn() &&
        request.resource.data.proId == request.auth.uid;

      // Either party may update (customer accepts; pro withdraws or edits).
      // Admins may update anything.
      allow update: if isSignedIn() && (
        resource.data.customerId == request.auth.uid ||
        resource.data.proId == request.auth.uid ||
        isAdmin()
      );

      // Only the pro who sent it, or an admin, may delete.
      allow delete: if isSignedIn() && (
        resource.data.proId == request.auth.uid ||
        isAdmin()
      );

    }


    // -------------------------------------------------------------------------
    // quoteRequests/{id}   (customer-initiated "please quote me" from Find Pros)
    // -------------------------------------------------------------------------

    match /quoteRequests/{reqId} {

      allow read: if isSignedIn() && (
        resource.data.customerId == request.auth.uid ||
        resource.data.proId == request.auth.uid ||
        isAdmin()
      );

      // Only customers may create; customerId must equal their uid.
      allow create: if isSignedIn() &&
        request.resource.data.customerId == request.auth.uid;

      // Only the receiving pro may respond (update status).
      // Admins may update anything.
      allow update: if isSignedIn() && (
        resource.data.proId == request.auth.uid ||
        isAdmin()
      );

      allow delete: if isAdmin();

    }


    // -------------------------------------------------------------------------
    // conversations/{id}
    // -------------------------------------------------------------------------

    match /conversations/{convId} {

      // Only participants may read the conversation header.
      allow read: if isSignedIn() &&
        request.auth.uid in resource.data.participants;

      // Any signed-in user may create a conversation, but their uid must
      // appear in the participants array.
      allow create: if isSignedIn() &&
        request.auth.uid in request.resource.data.participants;

      // Only participants may update (e.g. lastMessage preview field).
      allow update: if isSignedIn() &&
        request.auth.uid in resource.data.participants;

      allow delete: if isAdmin();


      // conversations/{id}/messages/{msgId}
      match /messages/{msgId} {

        // Only participants of the parent conversation may read messages.
        allow read: if isSignedIn() &&
          request.auth.uid in
            get(/databases/$(database)/documents/conversations/$(convId))
              .data.participants;

        // A participant may send a message; senderId must equal their uid.
        allow create: if isSignedIn() &&
          request.auth.uid in
            get(/databases/$(database)/documents/conversations/$(convId))
              .data.participants &&
          request.resource.data.senderId == request.auth.uid;

        // Messages are immutable once sent. Only admins may alter or remove them.
        allow update, delete: if isAdmin();

      }

    }


    // -------------------------------------------------------------------------
    // calendarEvents/{id}
    // -------------------------------------------------------------------------

    match /calendarEvents/{eventId} {

      // Only the owning user may read, update, or delete their events.
      allow read, update, delete: if isSignedIn() &&
        resource.data.userId == request.auth.uid;

      // On creation, the userId field must equal the creator's uid.
      allow create: if isSignedIn() &&
        request.resource.data.userId == request.auth.uid;

    }


    // -------------------------------------------------------------------------
    // reviews/{id}
    // -------------------------------------------------------------------------

    match /reviews/{reviewId} {

      // Reviews are public to all signed-in users (displayed on pro profiles).
      allow read: if isSignedIn();

      // Only the customer who wrote the review may create it;
      // their uid must be the customerId.
      allow create: if isSignedIn() &&
        request.resource.data.customerId == request.auth.uid;

      // Customers may edit their own review. Admins may edit any.
      allow update: if isSignedIn() && (
        resource.data.customerId == request.auth.uid ||
        isAdmin()
      );

      // Customers may delete their own review. Admins may delete any.
      allow delete: if isSignedIn() && (
        resource.data.customerId == request.auth.uid ||
        isAdmin()
      );

    }


    // -------------------------------------------------------------------------
    // notifications/{id}
    // -------------------------------------------------------------------------

    match /notifications/{notifId} {

      // Only the intended recipient may read their notifications.
      allow read: if isSignedIn() &&
        request.auth.uid == resource.data.userId;

      // Any signed-in user may create a notification for another user.
      // This is necessary because the client creates notifications as a side
      // effect of actions (e.g. sending a quote notifies the customer).
      // For tighter control, move notification creation to Cloud Functions.
      allow create: if isSignedIn();

      // Only the recipient may mark a notification as read.
      // No other fields may be changed.
      allow update: if isSignedIn() &&
        request.auth.uid == resource.data.userId &&
        request.resource.data.diff(resource.data)
          .affectedKeys().hasOnly(['read']);

      // Only the recipient or an admin may delete.
      allow delete: if isSignedIn() && (
        request.auth.uid == resource.data.userId ||
        isAdmin()
      );

    }


    // -------------------------------------------------------------------------
    // businessProfiles/{uid}   (document ID == the pro's uid)
    // -------------------------------------------------------------------------

    match /businessProfiles/{uid} {

      // Any signed-in user may read business profiles (Find Pros, Pro Profile).
      allow read: if isSignedIn();

      // Only the pro whose uid matches the document ID may create or update
      // their own business profile.
      allow create, update: if isOwner(uid);

      // The pro themselves or an admin may delete.
      allow delete: if isOwner(uid) || isAdmin();

    }


    // -------------------------------------------------------------------------
    // savedPros/{id}
    // -------------------------------------------------------------------------

    match /savedPros/{docId} {

      // Only the customer who saved the pro may read the record.
      allow read: if isSignedIn() &&
        resource.data.customerId == request.auth.uid;

      // Only a customer may create; their uid must be the customerId.
      allow create: if isSignedIn() &&
        request.resource.data.customerId == request.auth.uid;

      // Only the customer who saved the pro may remove the record.
      allow delete: if isSignedIn() &&
        resource.data.customerId == request.auth.uid;

      // The record is otherwise immutable (admins only for updates).
      allow update: if isAdmin();

    }


    // -------------------------------------------------------------------------
    // savedAddresses/{id}
    // -------------------------------------------------------------------------

    match /savedAddresses/{docId} {

      // Only the owning user may read, update, or delete their addresses.
      allow read, update, delete: if isSignedIn() &&
        resource.data.userId == request.auth.uid;

      // On creation, the userId field must equal the creator's uid.
      allow create: if isSignedIn() &&
        request.resource.data.userId == request.auth.uid;

    }


    // -------------------------------------------------------------------------
    // jobTemplates/{id}
    // -------------------------------------------------------------------------

    match /jobTemplates/{docId} {

      // Only the owning user may read, update, or delete their templates.
      allow read, update, delete: if isSignedIn() &&
        resource.data.userId == request.auth.uid;

      // On creation, the userId field must equal the creator's uid.
      allow create: if isSignedIn() &&
        request.resource.data.userId == request.auth.uid;

    }


    // -------------------------------------------------------------------------
    // estimates/{id}
    // -------------------------------------------------------------------------

    match /estimates/{docId} {

      // Only the pro who owns the estimate, or an admin, may read or update it.
      allow read, update, delete: if isSignedIn() && (
        resource.data.proId == request.auth.uid ||
        isAdmin()
      );

      // Only the pro may create an estimate; their uid must be the proId.
      allow create: if isSignedIn() &&
        request.resource.data.proId == request.auth.uid;

    }


    // -------------------------------------------------------------------------
    // customerNotes/{id}   (document ID == "${proId}_${customerId}")
    // -------------------------------------------------------------------------

    match /customerNotes/{docId} {

      // Only the pro who owns the note may read, update, or delete it.
      allow read, update, delete: if isSignedIn() &&
        resource.data.proId == request.auth.uid;

      // On creation, the proId must equal the creator's uid.
      allow create: if isSignedIn() &&
        request.resource.data.proId == request.auth.uid;

    }


    // -------------------------------------------------------------------------
    // declinedLeads/{id}
    // -------------------------------------------------------------------------

    match /declinedLeads/{docId} {

      // Only the pro who declined the lead may read or delete the record.
      allow read, delete: if isSignedIn() &&
        resource.data.proId == request.auth.uid;

      // On creation, the proId must equal the creator's uid.
      allow create: if isSignedIn() &&
        request.resource.data.proId == request.auth.uid;

    }


    // -------------------------------------------------------------------------
    // reports/{id}
    // -------------------------------------------------------------------------

    match /reports/{reportId} {

      // The reporter may read their own report. Admins may read all reports.
      allow read: if isSignedIn() && (
        resource.data.reporterId == request.auth.uid ||
        isAdmin()
      );

      // Any signed-in user may file a report; reporterId must equal their uid.
      allow create: if isSignedIn() &&
        request.resource.data.reporterId == request.auth.uid;

      // Only admins may change report status (reviewed / resolved).
      allow update: if isAdmin();

      // Only admins may delete reports.
      allow delete: if isAdmin();

    }


    // -------------------------------------------------------------------------
    // availability/{uid}   (document ID == the pro's uid)
    // -------------------------------------------------------------------------

    match /availability/{uid} {

      // Any signed-in user may read availability (needed for scheduling).
      allow read: if isSignedIn();

      // Only the pro whose uid matches the document ID may write their
      // own availability.
      allow write: if isOwner(uid);

    }

  }
}
